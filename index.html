create_tournaments_table.php

Schema::create('tournaments', function (Blueprint $table) {
  $table->id();
  $table->string('title');
  $table->dateTime('date');
  $table->string('Status')->default('active');
  $table->integer('idBoutiques');
  $table->decimal('price', 8, 2)->default(0);
  $table->integer('templix_reward')->default(0);
  $table->boolean('requires_payment')->default(false);
  $table->timestamps();
  $table->index(['date', 'Status']);
  $table->index('idBoutiques');
});

TournamentController

public function byCity($city) {
  $tournaments = Tournament::with(['boutique', 'preregistrations'])
    ->whereHas('boutique', function($query) use ($city) {
      $query->where('VilleRCS', $city);
    })
    ->where('Status', 'active')
    ->where('date', '>=', now())
    ->orderBy('date')
    ->get();

  return response()->json($tournaments);
}

// Validation et création
public function store(Request $request) {
  $validated = $request->validate([
    'title' => 'required|string|max:255',
    'game_type' => 'required|string|max:100',
    'date' => 'required|date',
  ]);

  Tournament::create($validated);
  return redirect()->route('admin.tournaments.index');
}

class TournamentRegistration {
  async submitRegistration() {
    const formData = new FormData(this.form);
    const response = await fetch('/api/preregistrations', {
      method: 'POST',
      headers: {
        'X-CSRF-TOKEN': csrfToken,
        'Accept': 'application/json'
      },
      body: formData
    });

    if (response.ok) {
      this.showSuccess('Inscription réussie !');
    } else {
      this.showError("Erreur d'inscription");
    }
  }
}

// Middleware AdminMiddleware
class AdminMiddleware {
  public function handle($request, Closure $next) {
    if (!auth()->check() || !auth()->user()->is_admin) {
      return redirect('/')->with('error', 'Acces refuse');
    }
    return $next($request);
  }
}

QuickRestore — logique principale

$availableTournaments = Tournament::where('Status', 'active')
  ->where('date', '>=', now())
  ->get();

foreach ($availableTournaments as $tournament) {
  // Tournoi complet ?
  if ($tournament->max_participants &&
      $tournament->preregistrations()
        ->where('idStatus', 1)->count()
      >= $tournament->max_participants) {
    $this->warn('Tournoi complet');

  Preregistration::create([

Problèmes Rencontrés & Solutions

Problèmes courants

Problème CSRF Token

  • Erreur 419 : Token CSRF manquant
  • Cause : Requêtes AJAX sans header
  • Solution : X-CSRF-TOKEN header
  • Résultat : API fonctionnelle

Problème Routes API

  • Erreur 404 : Endpoint non trouvé
  • Cause : URL mal configurée
  • Solution : Correction routes web.php
  • Résultat : Endpoints opérationnels

Problème Performance

  • N+1 queries : Requêtes multiples
  • Cause : Pas d'eager loading
  • Solution : with(['boutique', 'preregistrations'])
  • Résultat : Optimisation 80%

Exemples de code
